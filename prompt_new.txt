**File:** `app/src/client/index.ts`

```typescript
export const useCheckInMutation = () => {
  const queryClient = useQueryClient();
  const [lastCheckIn, setLastCheckIn] = useAtom($lastCheckIn); // Jotai

  return useMutation({
    mutationFn: postCheckIn,

    // 1. ON MUTATE (Run before request goes out)
    onMutate: async (newCheckIn) => {
      // A. Stop any outgoing refetches so they don't overwrite our optimistic update
      await queryClient.cancelQueries({ queryKey: ["gamification"] });

      // B. Snapshot the previous value (for rollback)
      const previousCheckIn = lastCheckIn;

      // C. Optimistically update Jotai
      if (lastCheckIn) {
        const estimatedTotal = lastCheckIn.xp_earned + newCheckIn.xp_amount;
        setLastCheckIn({
          ...lastCheckIn,
          xp_earned: estimatedTotal,
          goal_met: estimatedTotal >= 50, // Instant gratification!
          lessons_completed: lastCheckIn.lessons_completed + 1,
          // Note: We do NOT update streak optimistically as it's complex logic on server
        });
      }

      // Return context for rollback
      return { previousCheckIn };
    },

    // 2. ON ERROR (Request failed)
    onError: (err, newCheckIn, context) => {
      // Rollback to the snapshot
      if (context?.previousCheckIn) {
        setLastCheckIn(context.previousCheckIn);
      }
      // Show toast error
    },

    // 3. ON SETTLED (Request finished, success or fail)
    onSettled: () => {
      // Sync with server to be 100% sure
      queryClient.invalidateQueries({ queryKey: ["gamification"] });
    },
  });
};

```

```python
async def get_xp_history(days: int = 30, ...):
    today = date.today()
    start_date = today - timedelta(days=days - 1)
    
    # 1. Fetch raw data
    records = await session.execute(
        select(DailyProgress)
        .where(DailyProgress.user_id == user.id)
        .where(DailyProgress.date_key >= start_date)
    )
    data_map = {r.date_key: r for r in records.scalars()}
    
    # 2. Zero-fill loop
    history = []
    current = start_date
    while current <= today:
        key = current.isoformat() # "2026-02-12"
        if key in data_map:
            history.append(data_map[key])
        else:
            # Create a dummy object with 0 XP
            history.append(DailyProgress(date_key=key, xp_earned=0, goal_met=False))
        current += timedelta(days=1)
        
    return history

```

### TASK 2: OFFLINE QUEUE MODULE

**File:** `app/src/client/offline.ts`

Create a robust queue system.

```typescript
import AsyncStorage from "@react-native-async-storage/async-storage";
import NetInfo from "@react-native-community/netinfo";
import { onlineManager } from "@tanstack/react-query";
import { client } from "./index"; // Your axios instance

const QUEUE_KEY = "offline_mutation_queue";

interface QueuedMutation {
  id: string; // UUID
  endpoint: string;
  payload: any;
  createdAt: number;
  retryCount: number;
}

// 1. Add to Queue
export const queueMutation = async (endpoint: string, payload: any) => {
  const currentQueue = await getQueue();
  const newItem: QueuedMutation = {
    id: crypto.randomUUID(),
    endpoint,
    payload,
    createdAt: Date.now(),
    retryCount: 0,
  };
  await AsyncStorage.setItem(QUEUE_KEY, JSON.stringify([...currentQueue, newItem]));
  return newItem; // Return so we can log it
};

// 2. Sync Logic (The Replay Mechanism)
export const syncOfflineQueue = async () => {
  const state = await NetInfo.fetch();
  if (!state.isConnected) return; // Still offline

  const queue = await getQueue();
  if (queue.length === 0) return;

  console.log(`🔄 Syncing ${queue.length} offline mutations...`);
  
  const remainingQueue: QueuedMutation[] = [];

  for (const item of queue) {
    try {
      // Replay the request
      await client.post(item.endpoint, item.payload);
      console.log(`✅ Synced mutation ${item.id}`);
    } catch (error) {
      console.error(`❌ Failed to sync ${item.id}`, error);
      // Logic: Only keep if it's a network error (5xx or timeout), drop if 4xx (bad request)
      if (item.retryCount < 3) {
        remainingQueue.push({ ...item, retryCount: item.retryCount + 1 });
      }
    }
  }

  await AsyncStorage.setItem(QUEUE_KEY, JSON.stringify(remainingQueue));
};

// Helper to read queue
const getQueue = async (): Promise<QueuedMutation[]> => {
  const raw = await AsyncStorage.getItem(QUEUE_KEY);
  return raw ? JSON.parse(raw) : [];
};

```

```python
from fastapi import Header, HTTPException, status
from sqlalchemy import delete
from server.settings import settings

# Security: Only allow this if a secret key matches server config
@router.post("/demo/reset", status_code=200, include_in_schema=False)
async def demo_reset_user(
    current_user: User = Depends(get_current_user),
    session: AsyncSession = Depends(get_session),
    x_demo_key: str | None = Header(default=None, alias="X-Demo-Key")
):
    # 1. Security Check
    if not settings.DEMO_MODE_ENABLED:
        raise HTTPException(status.HTTP_403_FORBIDDEN, detail="Demo mode disabled")
    if x_demo_key != settings.DEMO_SECRET_KEY:
        raise HTTPException(status.HTTP_403_FORBIDDEN, detail="Invalid demo key")

    today = date.today()
    period_key = get_period_key(today)

    # 2. Nuclear Option (For this user only)
    # A. Delete today's progress log
    await session.execute(
        delete(DailyProgress)
        .where(DailyProgress.user_id == current_user.id)
        .where(DailyProgress.date_key == today)
    )

    # B. Reset Leaderboard Entry for this week
    await session.execute(
        delete(LeaderboardEntry)
        .where(LeaderboardEntry.user_id == current_user.id)
        .where(LeaderboardEntry.period_key == period_key)
    )

    # C. Reset Gamification Summary (Streak)
    # Note: We might want to set streak to 0, or keep it high for the demo?

    user_gamification = await session.get(UserGamification, current_user.id)
    if user_gamification:
        user_gamification.current_streak = 0
        user_gamification.last_activity_date = None
        session.add(user_gamification)

    await session.commit()
    return {"status": "reset_complete", "user": current_user.name}
```
